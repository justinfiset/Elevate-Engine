115 void Renderer::InitSSAORenderer(uint32_t width, uint32_t height)
116 {
117 s_ssaoShader = ShaderManager::LoadShader(
118 "ssao",
119 "engine://Shaders/SSAO.vert",
120 "engine://Shaders/SSAO.frag",
121 EE_SHADER_HEADER,
122 EE_SHADER_HEADER
123 );
124
125 s_ssaoBlurShader = ShaderManager::LoadShader(
126 "ssaoBlur",
127 "engine://Shaders/SSAOBlur.vert",
128 "engine://Shaders/SSAOBlur.frag",
129 EE_SHADER_HEADER,
130 EE_SHADER_HEADER
131 );
132
133 std::vector<glm::vec3> ssaoNoise;
134 std::uniform_real_distribution<float> randomFloats(0.0f, 1.0f);
135 std::default_random_engine generator;
136
137 for (uint32_t i = 0; i < 16; ++i)
138 {
139 glm::vec3 noise(
140 randomFloats(generator) * 2.0f - 1.0f,
141 randomFloats(generator) * 2.0f - 1.0f,
142 0.0f
143 );
144 ssaoNoise.push_back(glm::normalize(noise));
145 }
146
147 TextureMetadata noiseMeta = TextureMetadataBuilder()
148 .Name("AONoiseTexture")
149 .size(4, 4)
150 .Format(TextureFormat::RGB32F)
151 .Usage(TextureType::Diffuse)
152 .Source(TextureSource::Generated)
153 .State(TextureState::Ready)
154 .Filter(TextureFilter::Nearest, TextureFilter::Nearest)
155 .Wrap(TextureWrap::Repeat, TextureWrap::Repeat)
156 .Mipmaps(false)
157 .Build();
158
159 s_ssaoNoiseTexture = Texture::CreateFromData(ssaoNoise.data(), noiseMeta);
160
161 s_ssaoKernel.reserve(AO_KERNER_SAMPLE_COUNT);
162 for (int i = 0; i < AO_KERNER_SAMPLE_COUNT; i++)
163 {
164 glm::vec3 sample(
165 RandomFloat(-1.0f, 1.0f),
166 RandomFloat(-1.0f, 1.0f),
167 RandomFloat(0.0f, 1.0f)
168 );
169
170 float scale = static_cast<float>(i) / static_cast<float>(AO_KERNER_SAMPLE_COUNT);
171
172 scale = glm::mix(
173 0.1f,
174 1.0f,
175 scale * scale
176 );
177
178 sample *= scale;
179
180 s_ssaoKernel.push_back(sample);
181 }
182
183 // Create the Framebuffer
184 s_ssaoFramebuffer.reset(Framebuffer::Create(width, height, { TextureFormat::RGBA })); // todo R8 (red)
185 s_ssaoBlurFramebuffer.reset(Framebuffer::Create(width, height, { TextureFormat::RGBA })); // todo R8 (red)
186 }

256 void Renderer::BeginFrame(const ScenePtr scene, const Camera& cam)
257 {
258 InvalidateStateCache();
259
260 s_API->ClearTextureBindings();
261
262 s_data.CameraPosition = cam.gameObject->GetPosition();
263
264 s_data.View = cam.GenViewMatrix();
265 s_data.Projection = cam.GetProjectionMatrix();
266 s_data.ViewProj = s_data.Projection * s_data.View;
267 s_data.ActiveLighting = scene->GetSceneLighting();
268 auto skybox = scene->GetSkybox().lock();
269
270 constexpr std::array<float, SHADOW_CASCADE_COUNT + 1> cascadeSplits = { 0.0f, 0.05f, 0.15f, 0.40f, 1.0f };
271 float clipRange = cam.GetFar() - cam.GetNear();
272
273 for (int i = 0; i < SHADOW_CASCADE_COUNT; i++)
274 {
275 float currentNear = cascadeSplits[i];
276 float currentFar = cascadeSplits[i + 1];
277
278 auto corners = cam.CalculateFrustumCorners(currentNear, currentFar);
279 s_data.LightSpaceMatrices[i] = scene->GetSceneLighting()->GetDirectionalLightSpaceMatrix(corners);
280 s_data.CascadeSplitDepths[i] = cam.GetNear() + (currentFar * clipRange);
281 }
282
283 s_data.ActiveCubemap = skybox.get();
284 }

399 {
400 if (!s_isStateCacheValid || newState.CullMode != s_currentState.CullMode)
401 {
402 s_API->SetCullingState(newState.CullMode);

405 if (!s_isStateCacheValid || newState.DepthWrite != s_currentState.DepthWrite)
406 {
407 s_API->SetDepthWrittingState(newState.DepthWrite);

512 void Renderer::RenderShaowMaps()
513 {
514 if (!s_data.ActiveLighting)
515 {
516 return;
517 }
518
519 auto* dirLight = s_data.ActiveLighting->GetDirLight();
520 if (dirLight)
521 {
522 DirectionalShadowSettings settings = dirLight->m_shadowSettings;
523
524 BindShader(s_shadowShader);
525
526 // Rescale the framebuffer if the resolution was edited
527 if (s_directionalShadowMaps[0]->GetWidth() != settings.Resolution || s_directionalShadowMaps[0]->GetHeight() != settings.Resolution)
528 {
529 for (int i = 0; i < SHADOW_CASCADE_COUNT; i++)
530 {
531 s_directionalShadowMaps[i]->Rescale(settings.Resolution, settings.Resolution);
532 }
533 }
534
535 RenderState shadowState;
536 shadowState.CullMode = CullFace::Front;
537 shadowState.DepthTest = true;
538 shadowState.DepthWrite = true;
539 shadowState.BlendMode = BlendModeType::None;
540 PushRenderState(shadowState);
541
542 const auto& bucket = s_commands.GetBucket(RenderBucket::GBuffer);
543
544 for (int i = 0; i < SHADOW_CASCADE_COUNT; i++)
545 {
546 s_directionalShadowMaps[i]->Use();
547 ClearDepth();
548
549 s_shadowShader->SetUniformMatrix4fv("lightSpaceMatrix", s_data.LightSpaceMatrices[i]);
550
551 for (const auto& command : bucket)
552 {
553 s_shadowShader->SetModelMatrix(command.Transform);
554 DrawArray(command.m_VertexArray);
555 }
556
557 s_directionalShadowMaps[i]->Unbind();
558 }
559 }

588 {
589 RenderState lightingState;
590 lightingState.BlendMode = BlendModeType::None;
591 lightingState.CullMode = CullFace::None;
592 lightingState.DepthTest = false;
593 lightingState.DepthWrite = false;
594 PushRenderState(lightingState);
595
596 s_lightPassFramebuffer->ClearAndUse();
597
598 BindShader(s_lightPassShader);
599
600 ApplySystemUniforms(s_lightPassShader);
601
602 // Lighting configuration
603 if (s_data.ActiveLighting)
604 {
605 s_data.ActiveLighting->UploadToShader(s_lightPassShader);
606 }
607
608 // Shadows
609 if (s_directionalShadowMaps[0])
610 {
611 int shadowSlots[SHADOW_CASCADE_COUNT];
612 for (int i = 0; i < SHADOW_CASCADE_COUNT; i++)
613 {
614 uint8_t slot = SHADOW_MAP_SLOT + i;
615 BindTexture(s_directionalShadowMaps[i]->GetDepthTexture(), slot);
616 shadowSlots[i] = slot;
617 }
618
619 s_lightPassShader->SetUniform1iv("shadowMapArray", SHADOW_CASCADE_COUNT, shadowSlots);
620 }
621
622 // GBuffer
623 BindTexture(s_geometryFramebuffer->GetColorTexture(0), 0);
624 s_lightPassShader->SetUniform1i("g_Color", 0);
625 BindTexture(s_geometryFramebuffer->GetColorTexture(1), 1);
626 s_lightPassShader->SetUniform1i("g_Normal", 1);
627 BindTexture(s_geometryFramebuffer->GetColorTexture(2), 2);
628 s_lightPassShader->SetUniform1i("g_Material", 2);
629 BindTexture(s_geometryFramebuffer->GetDepthTexture(), 3);
630 s_lightPassShader->SetUniform1i("g_Depth", 3);
631 // SSAO
632 BindTexture(s_ssaoBlurFramebuffer->GetColorTexture(), 4);
633 s_lightPassShader->SetUniform1i("g_SSAO", 4);
634
635 DrawArray(s_fullscreenQuad.GetVertexArray());

692 void Renderer::RenderBloom()
693 {
694 RenderState state;
695
696 state.BlendMode = BlendModeType::None;
697 state.CullMode = CullFace::None;
698 state.DepthTest = false;
699 state.DepthWrite = false;
700 PushRenderState(state);
701
702 BindShader(s_bloomDownsampleShader);
703
704 TexturePtr currentSource = s_lightPassFramebuffer->GetColorTexture();
705
706 // Downsample trought the bloom mip chain
707 for (size_t i = 0; i < s_bloomMipChain.size(); i++)
708 {
709 auto& targetFramebuffer = s_bloomMipChain[i];
710 targetFramebuffer->ClearAndUse();
711
712 s_bloomDownsampleShader->SetUniform2f("u_ScreenRes", glm::vec2(currentSource->GetWidth(), currentSource->GetHeight()));
713 BindTexture(currentSource, 0);
714 s_bloomDownsampleShader->SetUniform1i("u_ScreenTex", 0);
715
716 s_bloomDownsampleShader->SetUniform1i("u_MipLevel", static_cast<int>(i));
717 s_bloomDownsampleShader->SetUniform1f("u_Threshold", 1.0f);
718 s_bloomDownsampleShader->SetUniform1f("u_SoftThreshold", 0.5f);
719
720 DrawArray(s_fullscreenQuad.GetVertexArray());
721 targetFramebuffer->Unbind();
722 currentSource = s_bloomMipChain[i]->GetColorTexture();
723 }
724
725 state.BlendMode = BlendModeType::Additive;
726 PushRenderState(state);
727
728 BindShader(s_bloomUpsampleShader);
729 s_bloomUpsampleShader->SetUniform1f("u_FilterRadius", 0.005f);
730
731 // Upsample the other way around
