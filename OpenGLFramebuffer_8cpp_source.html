117 {
118 if (m_colorAttachments.empty())
119 {
120 EE_CORE_WARN("Cannot blit a framebuffer without color attachments to screen!");
121 return;
122 }
123
124 GLCheck(glBindFramebuffer(GL_READ_FRAMEBUFFER, m_frameBufferId));
125 GLCheck(glBindFramebuffer(GL_DRAW_FRAMEBUFFER, 0));
126
127 uint32_t srcW = !m_colorAttachments.empty() ? m_colorAttachments[0].Texture->GetWidth() : 1280;
128 uint32_t srcH = !m_colorAttachments.empty() ? m_colorAttachments[0].Texture->GetHeight() : 720;
129
130 bool isMagnifying = (screenWidth >= srcW) && (screenHeight >= srcH);
131 auto filterType = isMagnifying
132 ? m_colorAttachments[0].Texture->GetMetadata().MagFilter
133 : m_colorAttachments[0].Texture->GetMetadata().MinFilter;
134 GLenum filter = ToOpenGL(filterType);
135 if (filter != GL_NEAREST && filter != GL_LINEAR)
136 {
137 filter = GL_LINEAR;
138 }
139
140 GLCheck(glBlitFramebuffer(
141 0, 0, srcW, srcH,
142 0, 0, screenWidth, screenHeight,
143 GL_COLOR_BUFFER_BIT,
144 filter
145 ));

147 GLCheck(glBindFramebuffer(GL_FRAMEBUFFER, 0));

150 void OpenGLFrameBuffer::Rescale(uint32_t width, uint32_t height)
151 {
152 m_width = width;
153 m_height = height;
154
155 if (width == 0 || height == 0) return;
156
157 Bind();
158
159 for (auto& attachment : m_colorAttachments)
160 {
161 if (auto& tex = attachment.Texture)

164
165 TextureMetadata meta = tex->GetMetadata();
166 meta.Width = width;
167 meta.Height = height;
168 tex->SetData(nullptr, meta);
169
170 GLuint texId = static_cast<GLuint>(reinterpret_cast<intptr_t>(tex->GetNativeHandle()));
171 GLCheck(glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0 + attachment.Index, GL_TEXTURE_2D, texId, 0));
172 }
173 }
174
175 if (m_depthAttachment.has_value())
176 {
177 if (m_depthAttachment->Texture)
178 {
179 auto& depthTex = m_depthAttachment->Texture;
180 TextureMetadata meta = depthTex->GetMetadata();
181 meta.Width = width;
182 meta.Height = height;
183 depthTex->SetData(nullptr, meta);
184
185 GLuint depthId = static_cast<GLuint>(reinterpret_cast<intptr_t>(depthTex->GetNativeHandle()));
186 GLenum attachmentType = (meta.Format == TextureFormat::DEPTHSTENCIL)
187 ? GL_DEPTH_STENCIL_ATTACHMENT
188 : GL_DEPTH_ATTACHMENT;
189
190 GLCheck(glFramebufferTexture2D(GL_FRAMEBUFFER, attachmentType, GL_TEXTURE_2D, depthId, 0));
191 }

194 GLCheck(glBindRenderbuffer(GL_RENDERBUFFER, m_renderBufferId));
