28 void SerializeRecursive(const PropertySet& fields, rapidjson::Value& parentObj, rapidjson::Document::AllocatorType& allocator)
29 {
30 for (const auto& property : fields)
31 {
32 rapidjson::Value key(property.Name.c_str(), allocator);
33
34 if (property.Type == EngineDataType::Array && std::holds_alternative<PropertyContainer>(property.Value))
35 {
36 const auto& container = std::get<PropertyContainer>(property.Value);
37 rapidjson::Value jsonArray(rapidjson::kArrayType);
38
39 for (const auto& elem : container.Children)

167 void ParseJsonValue(const rapidjson::Value& val, PropertyField& prop, const std::string& currentPath, uint16_t currentDepth)
168 {
169 if (val.IsBool())
170 {
171 prop.Type = EngineDataType::Bool;
172 prop.Value = val.GetBool();
173 }
174 else if (val.IsInt64() || val.IsInt() || val.IsUint() || val.IsUint64())
175 {
176 prop.Type = EngineDataType::Int;
177 prop.Value = static_cast<int64_t>(val.GetInt64());
178 }
179 else if (val.IsDouble())
180 {
181 prop.Type = EngineDataType::Double;
182 prop.Value = val.GetDouble();
183 }
184 else if (val.IsString())
185 {
186 const std::string s = val.GetString();
187
188 if (s == "NaN")
189 {
190 prop.Type = EngineDataType::Double;
191 prop.Value = std::numeric_limits<double>::quiet_NaN();
192 }
193 else if (s == "Infinity")
194 {
195 prop.Type = EngineDataType::Double;
196 prop.Value = std::numeric_limits<double>::infinity();
197 }
198 else if (s == "-Infinity")
199 {
200 prop.Type = EngineDataType::Double;
201 prop.Value = -std::numeric_limits<double>::infinity();
202 }
203 else
204 {
205 prop.Type = EngineDataType::String;
206 prop.Value = s;
207 }
208 }
209 else if (val.IsObject())
210 {
211 PropertyContainer container;
212 DeserializeRecursive(val, container.Children, currentPath, currentDepth + 1);
213 prop.Type = EngineDataType::Custom;
214 prop.Value = container;
215 }
216 else if (val.IsArray())
217 {
218 PropertyContainer container;
219 PropertySet arraySet;
220
221 for (rapidjson::SizeType i = 0; i < val.Size(); ++i)
222 {
223 const auto& elemVal = val[i];
224 PropertyField elemProp;
225 std::string indexStr = "[" + std::to_string(i) + "]";
226
227 elemProp.Name = indexStr;
228 elemProp.Path = currentPath + indexStr;
229 elemProp.Depth = currentDepth + 1;
230
231 ParseJsonValue(elemVal, elemProp, elemProp.Path, currentDepth + 1);
232 arraySet.push_back(elemProp);
233 }
234
235 container.Children = arraySet;
236 prop.Type = EngineDataType::Array;
237 prop.Value = container;
238 }
