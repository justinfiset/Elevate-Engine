192 void JoltPhysicsSystem::AddRigidbody(IPhysicsRigidbody* rigidbody)
193 {
194 JPH::StaticCompoundShapeSettings compoundSettings;
195 const glm::vec3 position = rigidbody->GetPosition();
196 const glm::quat quat = rigidbody->GetRotation();
197 const auto& colliders = rigidbody->GetPhysicsColliders();
198
199 if (colliders.empty())
200 {
201 return;
202 }
203
204 for (const IPhysicsCollider* collider : colliders)
205 {
206 auto shapeResult = JoltShapeFactory::Create(collider);
207
208 if (!shapeResult)
209 {
210 CoreLogger::Warn("JoltShapeFactory returned a non valid collider.");
211 return;
212 }
213
214 const auto& center = collider->GetCenter();
215 compoundSettings.AddShape(
216 JPH::Vec3(center.x, center.y, center.z),
217 JPH::Quat::sIdentity(),
218 shapeResult
219 );
220 }
221
222 JPH::ShapeSettings::ShapeResult compoundResult = compoundSettings.Create();
223
224 if (compoundResult.HasError())
225 {
226 CoreLogger::Error("Failed to create Jolt Physics shape: {}", compoundResult.GetError().c_str());
227 return;
228 }
229
230 JPH::BodyCreationSettings bodySettings(
231 compoundResult.Get(),
232 JPH::RVec3(position.x, position.y, position.z),
233 JPH::Quat(quat.x, quat.y, quat.z, quat.w),
234 GetBodyMotionType(rigidbody),
235 GetObjectLayer(rigidbody)
236 );
237
238 auto activation = GetBodyActivation(rigidbody);
239 auto bodyID = m_BodyInterface->CreateAndAddBody(bodySettings, activation);
240
241 if (bodyID.IsInvalid())
242 {
243 CoreLogger::Warn("Failed to create Jolt physics body!");
244 return;
245 }
246
247 JoltPhysicsBody joltBody;
248 joltBody.rigidbody = rigidbody;
249 joltBody.bodyID = bodyID;
250 m_Bodies.push_back(joltBody);
251 }

253 void JoltPhysicsSystem::RemoveRigidbody(IPhysicsRigidbody* rigidbody)
254 {
255 auto it = std::find_if(m_Bodies.begin(), m_Bodies.end(), [rigidbody](const auto& body)
256 {
257 return body.rigidbody == rigidbody;
258 });
259
260 if (it == m_Bodies.end())
261 {
262 return;
263 }
264
265 m_BodyInterface->RemoveBody(it->bodyID);
266 m_BodyInterface->DestroyBody(it->bodyID);
267
268 m_Bodies.erase(it);
269 }

277 void JoltPhysicsSystem::SyncDynamicBodies()
278 {
279 for (auto& body : m_Bodies)
280 {
281 IPhysicsRigidbody* rigidbody = body.rigidbody;
282
283 if (!rigidbody->IsDynamic())
284 {
285 continue;
286 }
287
288 // Fetch the data from jolt
289 JPH::Vec3 position;
290 JPH::Quat rotation;
291 m_BodyInterface->GetPositionAndRotation(body.bodyID, position, rotation);
292
293 // Apply the data to the world object through the physics interface.
294 rigidbody->SetPosition(glm::vec3{ position.GetX(), position.GetY(), position.GetZ() });
295 rigidbody->SetRotation(glm::quat{ rotation.GetX(), rotation.GetY(), rotation.GetZ(), rotation.GetW() });
296 }
