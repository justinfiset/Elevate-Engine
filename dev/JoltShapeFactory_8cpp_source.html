26 JPH::ShapeRefC JoltShapeFactory::Create(const IPhysicsCollider* collider)
27 {
28 if (!collider)
29 {
30 CoreLogger::Warn("Cannot create a shape out of a null collider.");
31 return nullptr;
32 }
33
34 auto* shape = collider->GetPhysicsShape();
35 if (!shape)
36 {
37 CoreLogger::Warn("Unsupported collider type.");
38 return nullptr;
39 }
40
41 if (const auto* boxShape = dynamic_cast<const BoxShape*>(shape))
42 {
43 return CreateBoxShape(*boxShape);
44 }
45
46 if (const auto* capsuleShape = dynamic_cast<const CapsuleShape*>(shape))
47 {
48 return CreateCapsuleShape(*capsuleShape);
49 }
50
51 if (const auto* sphereShape = dynamic_cast<const SphereShape*>(shape))
52 {
53 return CreateSphereShape(*sphereShape);
54 }
55
56 if (const auto* planeShape = dynamic_cast<const PlaneShape*>(shape))
57 {
58 return CreatePlaneShape(*planeShape);
59 }
60
61 CoreLogger::Warn("Unsupported collider shape.");
62 return nullptr;
63 }

75 JPH::ShapeRefC JoltShapeFactory::CreateBoxShape(const BoxShape& shape)
76 {
77 const auto& extents = shape.GetHalfExtents();
78 JPH::BoxShapeSettings shapeSettings(JPH::Vec3(extents.x, extents.y, extents.z));
79 JPH::ShapeSettings::ShapeResult shapeResult = shapeSettings.Create();
