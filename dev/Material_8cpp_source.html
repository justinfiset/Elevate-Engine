109 std::shared_ptr<Texture> Material::GetTextureForUniform(const std::string& uniformName) const
110 {
111 if (m_textures.empty()) return nullptr;
112
113 auto it = m_textures.find(uniformName);
114 if (it != m_textures.end()) return it->second;
115
116 static auto containsCaseless = [](std::string_view str, std::string_view sub) {
117 return std::search(str.begin(), str.end(), sub.begin(), sub.end(),
118 [](char a, char b) { return ::tolower(a) == ::tolower(b); }) != str.end();
119 };
120
121 static constexpr std::string_view keywords[] = { "diffuse", "specular", "ambient", "normal" };
122
123 for (const auto& [key, tex] : m_textures)
124 {
125 for (auto kw : keywords)
126 {
127 if (containsCaseless(uniformName, kw) && containsCaseless(key, kw))
128 return tex;
129 }
130 }
131
132 return nullptr;
133 }

135 void Material::Apply()
136 {
137 ResolveShader();
138
139 if (!m_shader)
140 {
141 return;
142 }
143
144 if (s_shaderBinder)
145 {
146 s_shaderBinder(m_shader);
147 }
148
149 const auto& layout = m_shader->GetLayout();
150
151 for (const auto& uniform : layout)
152 {
153 if (uniform.Type == ShaderDataType::Sampler2D) continue;
154
155 if (uniform.Name == ShaderUniforms::ViewProj ||
156 uniform.Name == ShaderUniforms::CamPos ||
157 uniform.Name == ShaderUniforms::Model)
158 {
159 continue;
160 }
161
162 if (m_definedUniforms[uniform.Index])
163 {
164 m_shader->SetUniform(uniform.Name, uniform.Type, m_buffer.data() + uniform.Offset);
165 }
166 }
167
168 uint32_t slot = 0;
169 std::string hasFlagName;
170 hasFlagName.reserve(64);
171
172 for (const auto& uniform : layout)
173 {
174 if (uniform.Type != ShaderDataType::Sampler2D) continue;
175
176 const std::string& texName = uniform.Name;
177 std::shared_ptr<Texture> texture = GetTextureForUniform(texName);
178
179 hasFlagName.assign("has_").append(texName);
180
181 if (texture && texture->IsTextureLoaded() && texture->GetWidth() > 0 && texture->GetHeight() > 0)
182 {
183 if (s_textureBinder)
184 {
185 s_textureBinder(texture, static_cast<uint8_t>(slot));
186 }
187 m_shader->SetUniform1i(texName, slot);
188 m_shader->SetUniform1i(hasFlagName, 1);
189 }
190 else
191 {
192 if (s_textureBinder)
193 {
194 s_textureBinder(TextureManager::GetDefaultTexture(), static_cast<uint8_t>(slot));
195 }
196 m_shader->SetUniform1i(texName, slot);
197 m_shader->SetUniform1i(hasFlagName, 0);
198 }
199
200 slot++;
201 }
202 }
