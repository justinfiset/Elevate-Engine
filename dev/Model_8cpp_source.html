102 {
103 auto resolvedPath = PathResolver::Resolve(path);
104 auto scheduler = Application::Get().GetServices().Resolve<ITaskScheduler>();
105 std::weak_ptr<ITaskScheduler> weakScheduler = scheduler;
106
107 scheduler->Submit(
108 TaskDescriptor(TaskAffinity::Any, TaskPriority::Normal, "Model Loading"),

133 bool Model::ParseModel(const std::filesystem::path& path, MeshData& data)
134 {
135 Assimp::Importer import;
136 const aiScene* scene = import.ReadFile(path.string(),
137 aiProcess_Triangulate | aiProcess_GenNormals | aiProcess_OptimizeMeshes |
138 aiProcess_ImproveCacheLocality | aiProcess_CalcTangentSpace);
139
140 if (!scene || scene->mFlags & AI_SCENE_FLAGS_INCOMPLETE || !scene->mRootNode)
141 {
142 CoreLogger::Error("ASSIMP LOADING ERROR : {}", import.GetErrorString());
143 return false;
144 }
145
146 ProcessNode(path.parent_path().string(), scene->mRootNode, scene, data);
147
148 if (data.Vertices.empty() || data.Indices.empty())
149 {
150 CoreLogger::Error("Model '{}' contains no geometry.", path.string());
151 return false;
152 }
153 return true;
154 }

156 void Model::ProcessNode(std::string basePath, aiNode* node, const aiScene* scene, MeshData& data)
157 {
158 // process all the node's MESHES (if any)
159 for (unsigned int i = 0; i < node->mNumMeshes; i++)
160 {
161 aiMesh* mesh = scene->mMeshes[node->mMeshes[i]];
162 ProcessMesh(basePath, mesh, scene, data);
163 }
164 // then do the same for each of its CHILDREN(S)
165 for (unsigned int i = 0; i < node->mNumChildren; i++)
166 {
167 ProcessNode(basePath, node->mChildren[i], scene, data); // Continue till we run out of childrens
168 }

171 void Model::ProcessMesh(std::string basePath, aiMesh* mesh, const aiScene* scene, MeshData& data)
172 {
173 uint32_t offset = (uint32_t)data.Vertices.size();
174
175 // Process vertices
176 for (unsigned int i = 0; i < mesh->mNumVertices; i++)
177 {
178 Vertex vertex;
179 ExtractMeshVertex(mesh, vertex, i);
180 data.Vertices.emplace_back(vertex);
181 }
182
183 // process indices
184 for (unsigned int i = 0; i < mesh->mNumFaces; i++)
185 {
186 aiFace face = mesh->mFaces[i];
187 for (unsigned int j = 0; j < face.mNumIndices; j++)
188 {
189 data.Indices.emplace_back(face.mIndices[j] + offset);
190 }
191 }
192
193 // process material
194 if (mesh->mMaterialIndex >= 0)
195 {
196 aiMaterial* material = scene->mMaterials[mesh->mMaterialIndex];
197
198 LoadMaterialTextures(basePath, material, aiTextureType_DIFFUSE, TextureType::Diffuse, data);
199 LoadMaterialTextures(basePath, material, aiTextureType_SPECULAR, TextureType::Specular, data);
200
201 // Load ambiant occlusion
202 LoadMaterialTextures(basePath, material, aiTextureType_AMBIENT_OCCLUSION, TextureType::AmbientOcclusion, data);
203 LoadMaterialTextures(basePath, material, aiTextureType_LIGHTMAP, TextureType::AmbientOcclusion, data);
204 LoadMaterialTextures(basePath, material, aiTextureType_AMBIENT, TextureType::AmbientOcclusion, data);
205
206 // Normal map
207 LoadMaterialTextures(basePath, material, aiTextureType_NORMALS, TextureType::Normal, data);
208 LoadMaterialTextures(basePath, material, aiTextureType_HEIGHT, TextureType::Normal, data);
209 }

212 void Model::ExtractMeshVertex(aiMesh* mesh, Vertex& vertex, int i)
213 {
214 // process vertex positions, normals and texture coordinates
215 glm::vec3 vector;
216 vector.x = mesh->mVertices[i].x;
217 vector.y = mesh->mVertices[i].y;
218 vector.z = mesh->mVertices[i].z;
219 vertex.Position = vector;
220
221 vector.x = mesh->mNormals[i].x;
222 vector.y = mesh->mNormals[i].y;
223 vector.z = mesh->mNormals[i].z;
224 vertex.Normal = vector;
225
226 if (mesh->mTextureCoords[0]) // does the mesh contain texture coordinates?
227 {
228 // text coords
229 glm::vec2 coords;
230 coords.x = mesh->mTextureCoords[0][i].x;
231 coords.y = mesh->mTextureCoords[0][i].y;
232 vertex.TexCoords = coords;
233 }
234
235 if (mesh->mTangents)
236 {
237 // tangent
238 vector.x = mesh->mTangents[i].x;
239 vector.y = mesh->mTangents[i].y;
240 vector.z = mesh->mTangents[i].z;
241 vertex.Tangent = vector;
242 }
243
244 if (mesh->mBitangents)
245 {
246 // bitangent
247 vector.x = mesh->mBitangents[i].x;
248 vector.y = mesh->mBitangents[i].y;
249 vector.z = mesh->mBitangents[i].z;
250 vertex.Bitangent = vector;
251 }

254 void Model::LoadMaterialTextures(std::string basePath, aiMaterial* mat, unsigned int type, TextureType texType, MeshData& data)
255 {
256 const auto textureType = static_cast<aiTextureType>(type);
257 for (unsigned int i = 0; i < mat->GetTextureCount(textureType); i++)
258 {
259 aiString str;
260 mat->GetTexture(textureType, i, &str);
261
262 std::filesystem::path texturePath(str.C_Str());
263 std::error_code existsError;
264 std::string path;
265 if (texturePath.is_absolute() || std::filesystem::exists(texturePath, existsError))
266 {
267 path = texturePath.lexically_normal().string();
268 }
269 else
270 {
271 path = (std::filesystem::path(basePath) / texturePath).lexically_normal().string();
272 }
273
274 bool skip = false;
275 for (std::shared_ptr<Texture> tex : data.Textures)
276 {
277 if (tex->MatchesPath(path))
278 {
279 skip = true;
280 break;
281 }
282 }
283
284 if (!skip)
285 {
286 data.Textures.emplace_back(Texture::CreateFromFile(path, texType));
287 }
288 }
