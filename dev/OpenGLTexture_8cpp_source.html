210 void OpenGLTexture::SetDataImpl(const void* data)
211 {
212 EnsureCreated();
213 Bind();
214
215 GLCheck(glPixelStorei(GL_UNPACK_ALIGNMENT, 1));
216
217 GLCheck(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, ToOpenGL(m_meta.WrapS)));
218 GLCheck(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, ToOpenGL(m_meta.WrapT)));
219 GLCheck(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GetMinFilter(m_meta.MinFilter, m_meta.Mipmaps)));
220 GLCheck(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, ToOpenGL(m_meta.MagFilter)));
221
222 if (m_meta.Format == TextureFormat::DEPTH ||
223 m_meta.Format == TextureFormat::DEPTH16 ||
224 m_meta.Format == TextureFormat::DEPTH24 ||
225 m_meta.Format == TextureFormat::DEPTH32F)
226 {
227#if defined(EE_PLATFORM_WEB)
228 GLCheck(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE));
229 GLCheck(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE));
230#else
231 constexpr float borderColor[] = { 1.0f, 1.0f, 1.0f, 1.0f };
232 GLCheck(glTexParameterfv(GL_TEXTURE_2D, GL_TEXTURE_BORDER_COLOR, borderColor));
233#endif
234 }
235
236 if (m_meta.Usage == TextureType::ShadowMap)
237 {
238 GLCheck(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_COMPARE_MODE, GL_COMPARE_REF_TO_TEXTURE));
239 GLCheck(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_COMPARE_FUNC, GL_LEQUAL));
240 }
241
242#ifdef EE_SUPPORTS_DSA
243 if (m_meta.Format == TextureFormat::GRAYSCALE)
244 {
245 GLint swizzleMask[] = { GL_RED, GL_RED, GL_RED, GL_ONE };
246 GLCheck(glTexParameteriv(GL_TEXTURE_2D, GL_TEXTURE_SWIZZLE_RGBA, swizzleMask));
247 }
248#endif
249
250 CoreLogger::Info("Creating Texture: ID={}, Size={}x{}, Format={}", m_textureID, m_meta.Width, m_meta.Height, (int)m_meta.Format);
251
252 uint32_t width = m_meta.Width > 0 ? m_meta.Width : 1;
253 uint32_t height = m_meta.Height > 0 ? m_meta.Height : 1;
254
255 const void* pixelsToUpload = data;
256
257 GLCheck(glTexImage2D(
258 GL_TEXTURE_2D,
259 0,
260 ToInternalFormat(m_meta.Format),
261 width,
262 height,
263 0,
264 ToOpenGL(m_meta.Format),
265 ToOpenGLType(m_meta.Format),
266 pixelsToUpload));
267
268 if (m_meta.Mipmaps && width > 1 && height > 1 && pixelsToUpload != nullptr)
269 {
270 GLCheck(glGenerateMipmap(GL_TEXTURE_2D));
271 }
272 }
