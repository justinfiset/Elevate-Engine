23 template <class ValueType, class StoredType>
24 ValueCodec MakeScalarCodec()
25 {
26 return {
27 // Memory -> PropertyField
28 [](const void *data, PropertyField &out)
29 {
30 out.Value = static_cast<StoredType>(*static_cast<const ValueType *>(data));
31 },
32 // PropertyField -> Memory
33 [](void *data, const PropertyField &in)

41 }

54 template <class VectorType, class ElementType, class StoredType, size_t Count, EngineDataType DataType>
55 ValueCodec MakeGlmVecCodec()
56 {
57 return {
58 // Memory -> PropertyField
59 [](const void *data, PropertyField &out)
60 {
61 const ElementType *source = glm::value_ptr(*static_cast<const VectorType *>(data));
62
63 PropertySet comps;
64 comps.reserve(Count);
65
66 for (size_t i = 0; i < Count; ++i)
67 {
68 PropertyField compField;
69 compField.Name = ComponentName(i, Count);
70 compField.Path = out.Path + "/" + compField.Name;
71 compField.Type = DataType;
72 compField.Depth = out.Depth + 1;
73 compField.Value = static_cast<StoredType>(source[i]);
74
75 comps.push_back(std::move(compField));
76 }
77
78 out.Value = PropertyContainer{std::move(comps)};
79 },
80 // Memory -> PropertyField
81 [](void *data, const PropertyField &in)
82 {
83 const auto *container = std::get_if<PropertyContainer>(&in.Value);
84 if (!container)
85 {
86 CoreLogger::Error("Failed to apply value codec: PropertyField does not contain a PropertyContainer.");
87 return false;
88 }
89
90 ElementType *destination = glm::value_ptr(*static_cast<VectorType *>(data));
91
92 if (container->Children.size() < Count)
93 {
94 CoreLogger::Error("ValueCodec: '{}' has {} components, expected {}.", in.Path, container->Children.size(), Count);
95 return false;
96 }
97
98 for (size_t i = 0; i < Count; ++i)
99 {
100 if (const auto *value = std::get_if<StoredType>(&container->Children.at(i).Value))
101 {
102 destination[i] = static_cast<ElementType>(*value);
103 }
104 else
105 {
106 CoreLogger::Error("Failed to apply value codec: PropertyField component does not contain the expected type.");
107 return false;
108 }
109 }
110
111 return true;
112 }};
113 }

218 std::array<ValueCodec, EngineDataTypeCount> BuildTable()
219 {
220 std::array<ValueCodec, EngineDataTypeCount> table{};
221
222 auto set = [&](EngineDataType type, ValueCodec codec)
223 {
224 table[static_cast<size_t>(type)] = std::move(codec);
225 };
226
227 set(EngineDataType::Bool, MakeScalarCodec<bool, bool>());
228 set(EngineDataType::Double, MakeScalarCodec<double, double>());
229 set(EngineDataType::String, MakeScalarCodec<std::string, std::string>());
230
231 set(EngineDataType::Int, MakeScalarCodec<int32_t, int64_t>());
232 set(EngineDataType::Int2, MakeIntVecCodec<2>());
233 set(EngineDataType::Int3, MakeIntVecCodec<3>());
234 set(EngineDataType::Int4, MakeIntVecCodec<4>());
235
236 set(EngineDataType::Float, MakeScalarCodec<float, double>());
237 set(EngineDataType::Float2, MakeFloatVecCodec<2>());
238 set(EngineDataType::Float3, MakeFloatVecCodec<3>());
239 set(EngineDataType::Float4, MakeFloatVecCodec<4>());
240
241 set(EngineDataType::Mat3, MakeGlmVecCodec<glm::mat3, float, double, 9, EngineDataType::Float>());
242 set(EngineDataType::Mat4, MakeGlmVecCodec<glm::mat4, float, double, 16, EngineDataType::Float>());
243
244 set(EngineDataType::GUID, MakeGuidCoded());
245 set(EngineDataType::Enum, MakeEnumCodec());
246 set(EngineDataType::ObjectPtr, MakeObjectPtrCodec());
247
248 return table;
249 }

251 const ValueCodec *ValueCodecRegistry::Find(EngineDataType type)
252 {
253 static const auto table = BuildTable();
254
255 const size_t index = static_cast<size_t>(type);

259 return nullptr;
