85std::array<glm::vec3, 8> Elevate::Camera::CalculateFrustumCorners(float nearScale, float farScale) const
86{
87 std::array<glm::vec3, 8> corners;
88
89 const glm::mat4 inverseViewProj = glm::inverse(GenViewProjectionMatrix());
90
91 std::array<glm::vec3, 8> ndcCorners;
92 ndcCorners[0] = { -1.f, -1.f, -1.f };
93 ndcCorners[1] = { 1.f, -1.f, -1.f };
94 ndcCorners[2] = { 1.f, 1.f, -1.f };
95 ndcCorners[3] = { -1.f, 1.f, -1.f };
96 ndcCorners[4] = { -1.f, -1.f, 1.f };
97 ndcCorners[5] = { 1.f, -1.f, 1.f };
98 ndcCorners[6] = { 1.f, 1.f, 1.f };
99 ndcCorners[7] = { -1.f, 1.f, 1.f };
100
101 for (std::size_t i = 0; i < ndcCorners.size(); ++i)
102 {
103 glm::vec4 worldCorner = inverseViewProj * glm::vec4(ndcCorners[i], 1.0f);
104 worldCorner /= worldCorner.w;
105
106 corners[i] = glm::vec3(worldCorner);
107 }
108
109 for (std::size_t i = 0; i < 4; ++i)
110 {
111 const glm::vec3 nearCorner = corners[i];
112 const glm::vec3 farCorner = corners[i + 4];
113
114 const glm::vec3 direction = farCorner - nearCorner;
115
116 corners[i] = nearCorner + direction * nearScale;
117 corners[i + 4] = nearCorner + direction * farScale;
118 }
119
120 return corners;
121}
