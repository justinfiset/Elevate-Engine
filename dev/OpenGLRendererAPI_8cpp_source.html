79 void OpenGLRendererAPI::DrawArray(const VertexArray* vao, DrawPrimitiveType primitive) const
80 {
81 if (!glfwGetCurrentContext())
82 {
83 CoreLogger::Error("No OpenGL context active!");
84 return;
85 }
86
87 if (!vao)
88 {
89 CoreLogger::Error("VAO or IndexBuffer is null!");
90 return;
91 }
92
93 vao->Bind();
94
95 if (vao->GetIndexBuffer())
96 {
97 if (vao->GetIndexBuffer()->GetCount() == 0)
98 {
99 CoreLogger::Error("IndexBuffer has 0 indices!");
100 return;
101 }
102
103 GLCheck(glDrawElements(DrawPrimitiveTypeToOpenGL(primitive), vao->GetIndexBuffer()->GetCount(), GL_UNSIGNED_INT, nullptr));
104 }
105 else
106 {
107 for (const auto& vbo : vao->GetVertexBuffers())
108 {
109 uint32_t vertexCount = vbo->GetSize() / vbo->GetLayout().GetStride();
110 GLCheck(glDrawArrays(DrawPrimitiveTypeToOpenGL(primitive), 0, vertexCount));
111 }
112 }
113 }
