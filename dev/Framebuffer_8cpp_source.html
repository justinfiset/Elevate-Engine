37 Framebuffer* Framebuffer::Create(uint32_t width, uint32_t height, std::vector<TextureFormat> colorFormats, bool depthAsRenderbuffer, TextureType depthUsage)
38 {
39 if (width == 0 || height == 0)
40 {
41 CoreLogger::Warn("Framebuffer::Create - Attempted to create framebuffer with 0x0 dimensions. Skipping.");
42 return nullptr;
43 }
44
45 size_t colorAttachmentCount = colorFormats.size();
46
47 std::vector<std::shared_ptr<Texture>> colorTextures;
48 colorTextures.reserve(colorAttachmentCount);
49 for (uint32_t i = 0; i < colorAttachmentCount; i++)
50 {
51 colorTextures.push_back(CreateColorTexture(width, height, colorFormats[i]));
52 }
53
54 std::shared_ptr<Texture> depthTex = nullptr;
55 if (!depthAsRenderbuffer)
56 {
57 depthTex = CreateDepthTexture(width, height, TextureFormat::DEPTH, depthUsage);
58 }
59 Framebuffer* buffer = Framebuffer::Create(colorTextures, depthTex, depthAsRenderbuffer);
60 buffer->m_width = width;
61 buffer->m_height = height;
62 return buffer;
63 }
