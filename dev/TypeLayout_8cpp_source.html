84 {

107 PropertySet CaptureArray(const TypeField& array, const std::string& path, uint16_t depth)
108 {
109 PropertySet set;
110

134 PropertyField CaptureField(const TypeField& field, const std::string& path, uint16_t depth)
135 {
136 PropertyField prop;
137 prop.Name = field.name;
138 prop.Type = field.type;
139 prop.Path = path;
140 prop.Depth = depth;
141 prop.Flags = GetFieldFlags(field);
142
143 if (field.type == EngineDataType::Array)
144 {
145 prop.Value = PropertyContainer{ CaptureArray(field, path, depth + 1) };
146 }
147 else if (!field.children.empty())
148 {
149 prop.Value = PropertyContainer{ CaptureChildren(field, path, depth + 1) };
150 }
151 else if (const ValueCodec* codec = ValueCodecRegistry::Find(field.type); codec && field.data)
152 {
153 codec->Capture(field.data, prop);
154 }
155 else
156 {
157 CoreLogger::Warn("CaptureState: no codec for '{}' (type {}, children {})",
158 path, static_cast<int>(field.type), field.children.size());
159 }
160
161 return prop;

200 void ApplyArray(const TypeField& array, const PropertyField& prop)
201 {
202 const auto* container = std::get_if<PropertyContainer>(&prop.Value);
203 if (!container || !array.data)
204 {
205 return;
206 }
207
208 if (array.ResizeArray)
209 {
210 array.ResizeArray(const_cast<void*>(array.data), container->Children.size());
211 }
212
213 const size_t count = array.GetArraySize ? array.GetArraySize(array.data) : 0;
214 for (size_t i = 0; i < container->Children.size() && i < count; ++i)
215 {
216 const PropertyField& elementProp = container->Children[i];
217
218 const void* elementData = array.GetElementAddress ? array.GetElementAddress(array.data, i) : nullptr;
219 if (!elementData)
220 {
221 continue;
222 }
223
224 const TypeField element = MakeArrayElementField(array, elementProp.Name, elementData);
225 ApplyField(element, elementProp, elementProp.Path);
226 }

229 void ApplyField(const TypeField& field, const PropertyField& prop, const std::string& path)
230 {
231 if (field.type == EngineDataType::Array)
232 {
233 ApplyArray(field, prop);
234 }
235 else if (!field.children.empty())
236 {
237 if (const auto* container = std::get_if<PropertyContainer>(&prop.Value))
238 {
239 ApplyFields(field.children, field.data, container->Children, path);
240 }
241 else
242 {
243 CoreLogger::Warn("ApplyState: '{}' should hold a container", path);
244 }
245 }
246 else if (const ValueCodec* codec = ValueCodecRegistry::Find(field.type); codec && field.data)
247 {
248 if (!codec->Apply(const_cast<void*>(field.data), prop))
249 {
250 CoreLogger::Warn("ApplyState: failed to restore '{}' (type {})", path, static_cast<int>(field.type));
251 }
252 }
253 else
254 {
255 CoreLogger::Warn("ApplyState: no codec for '{}' (type {}, children {})",
256 path, static_cast<int>(field.type), field.children.size());
257 }
258 }
