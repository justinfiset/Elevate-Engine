49static void Elevate_MouseButtonCallback(GLFWwindow* window, int button, int action, int mods)
50{
51 auto* dataPtr = static_cast<Elevate::WindowData*>(glfwGetWindowUserPointer(window));
52 if (!dataPtr)
53 {
54 return;
55 }
56
57 const auto mouseButton = static_cast<Elevate::MouseButton>(button);
58
59 if (action == GLFW_PRESS)
60 {
61 Elevate::MouseButtonPressedEvent event(mouseButton);
62 dataPtr->EventCallback(event);
63 }
64 else if (action == GLFW_RELEASE)
65 {
66 Elevate::MouseButtonReleasedEvent event(mouseButton);
67 dataPtr->EventCallback(event);
68 }
69}
