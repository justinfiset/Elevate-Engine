87 void EditorLayer::OnAttach()
88 {
89 SetDarkTheme();
90
91 m_sceneManager = Application::Get().GetServices().Resolve<SceneManager>();
92 m_EditorScene = m_sceneManager->CreateScene("Editor", SceneType::EditorScene);
93
94 // Editor Camera
95 m_CameraObject = GameObject::Create("Editor Camera", m_EditorScene);
96 m_CameraObject->AddComponent<EditorCamera>(60.0f);
97
98 // Setup the grid shader ///////////////////////////
99 m_GridShader = Shader::CreateFromFiles(
100 "editor://Shaders/grid.vert",
101 "editor://Shaders/grid.frag"
102 );
103
104 MaterialPtr gridMaterial = MaterialRegistry::LoadMaterial(m_GridShader);
105 gridMaterial->Set<glm::vec4>("lineColor", { 0.9, 0.9, 0.9, 0.5 });
106 gridMaterial->Set<glm::vec4>("backgroundColor", { 0.6, 0.6, 0.6, 0.025 });
108
109 // Grid
110 m_GridObject = GameObject::Create("Editor Grid", m_EditorScene);
111 Model& gridModel = m_GridObject->AddComponent<Model>(PrimitiveType::Plane);
112
113 RenderState gridState;
114 gridState.BlendMode = BlendModeType::Alpha;
115 gridState.CullMode = CullFace::None;
116 gridState.DepthTest = true;
117 gridState.DepthWrite = false;
118 gridMaterial->SetRenderState(gridState);
119 gridMaterial->SetBucket(RenderBucketType::Editor);
120 gridModel.SetMaterial(gridMaterial);
121 m_GridObject->SetScale({ 50, 50, 50 });
122 }

301 void EditorLayer::OnEvent(Event& event)
302 {
303 switch (event.GetEventType())
304 {
305 case EventType::KeyPressed:
306 {
307 KeyEvent& ke = (KeyEvent&)event;
308
309 if (ke.GetKeyCode() == KeyCode::Delete)
310 {
311 if (auto lockedObj = m_SelectedObject.lock())
312 {
313 if (auto gameObj = std::dynamic_pointer_cast<GameObject>(lockedObj))
314 {
315 Execute(std::make_unique<DeleteGameobjectCommand>(gameObj));
316 }
317 }
318 }
319
320 if (Input::IsKeyPressed(KeyCode::LeftControl))
321 {
322 if (ke.GetKeyCode() == KeyCode::Z)
323 {
324 Undo();
325 }
326 else if (ke.GetKeyCode() == KeyCode::Y)
327 {
328 Redo();
329 }
330 }
331 break;
332 }
333 default:
334 break;
335 }
336
337 if (m_EditorScene)
338 {
339 m_EditorScene->Notify(event);
340 }
341 }
