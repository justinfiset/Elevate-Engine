102 OpenGLIndexBuffer::OpenGLIndexBuffer(const void* indices, uint32_t count)
103 : m_count(count)
104 {
105 GLCheck(CreateBuffers(1, &m_rendererID));
106#ifdef EE_SUPPORTS_DSA
107 GLCheck(glNamedBufferData(m_rendererID, count * sizeof(uint32_t), indices, GL_STATIC_DRAW));
108#else
109 // ELEMENT_ARRAY_BUFFER bindings are VAO state. Creating an index buffer must
110 // not replace the element binding of whichever VAO happens to be bound,
111 // otherwise publishing a mesh can corrupt the fullscreen quad.
112 GLCheck(glBindVertexArray(0));
113 GLCheck(glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, m_rendererID));
114 GLCheck(glBufferData(GL_ELEMENT_ARRAY_BUFFER, count * sizeof(uint32_t), indices, GL_STATIC_DRAW));
115 GLCheck(glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0));
116#endif
117 }
