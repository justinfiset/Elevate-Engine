101 // Event Dispatcher
102 void Application::OnEvent(Event& e)
103 {
104 EventDispatcher dispatcher(e);
105 dispatcher.Dispatch<WindowCloseEvent>(BIND_EVENT_FN(OnWindowClose));
106 dispatcher.Dispatch<WindowFocusEvent>(BIND_EVENT_FN(OnWindowFocusEvent));
107 dispatcher.Dispatch<FramebufferResizeEvent>(BIND_EVENT_FN(OnFramebufferResize));
108
109 // Keyboard
110 dispatcher.Dispatch<KeyPressedEvent>(BIND_EVENT_FN(OnKeyPressedEvent));
111 dispatcher.Dispatch<KeyReleasedEvent>(BIND_EVENT_FN(OnKeyReleasedEvent));
112
113 // Mousse
114 dispatcher.Dispatch<MouseButtonPressedEvent>(BIND_EVENT_FN(OnMouseButtonPressedEvent));
115 dispatcher.Dispatch<MouseButtonReleasedEvent>(BIND_EVENT_FN(OnMouseButtonReleasedEvent));
116
117 for (auto it = m_LayerStack.end(); it != m_LayerStack.begin();)
118 {
119 (*--it)->OnEvent(e);
120 if (e.Handled)
121 break;
122 }
123 }

181 void Application::EngineFrame(bool skipEventPolling) // Each frame
182 {
183 static float lastTime = 0.0f;
184
185#if !EE_ASSERTS_ENABLED
186 try
187 {
188#endif // #if EE_ASSERTS_ENABLED
189 // TIME UPDATE //////////////////
190 float currentTime = Time::GetCurrentTime();
191 float rawDeltaTime = currentTime - lastTime;
192 float deltaTime = std::min(rawDeltaTime, 0.25f); // Clamping to prevent weird behaviours
193 Time::Update((float)m_Window->GetTime(), deltaTime);
194 lastTime = currentTime;
196
197 m_Scheduler->Update();
198
199 SoundEngine::RenderAudio();
200 m_PhysicsSystem->Update(Time::GetDeltaTime());
201 TextureManager::UpdateLoadingTextures();
202
203 for (Layer* layer : m_LayerStack)
204 {
205 layer->OnUpdate();
206 }
207
208 // Draw Layers and Scenes
209 for (Layer* layer : m_LayerStack)
210 {
211 layer->OnRender();
212 }
213
214 Renderer::RenderFrame();
215
216 // If not in the editor, dump the content of the main color buffer to the main buffer
217#ifndef EE_EDITOR_BUILD
218 Renderer::Present(m_Window->GetWidth(), m_Window->GetHeight());
219#endif
220
221 // imgui
222 for (Layer* layer : m_LayerStack)
223 {
224 layer->OnImGuiRenderBegin();
225 }
226
227 for (Layer* layer : m_LayerStack)
228 {
229 layer->OnImGuiRender();
230 }
231
232 for (Layer* layer : m_LayerStack)
233 {
234 layer->OnImGuiRenderEnd();
235 }
236
237 Renderer::InvalidateStateCache(); // Clear the renderer state after ImGui rendering
238 Input::ManageMidStates(); // Manage Key/Button up and down state
239
240 Renderer::FlushBuffers();
241 // Poll events and swap buffers
242 m_Window->OnUpdate(skipEventPolling);
243
244#ifdef EE_PLATFORM_WEB
245 if (!m_Running)
246 {
247 emscripten_cancel_main_loop();
248 Exit();
249 }
250#endif
251
252#if !EE_ASSERTS_ENABLED
253 }
254 catch (const std::exception& exc)
255 {
256 CoreLogger::Error("{}", exc.what());
257 }
258#endif // #if EE_ASSERTS_ENABLED
259 }
