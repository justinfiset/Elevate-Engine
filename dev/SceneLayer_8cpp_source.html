60 const auto type = event.GetEventType();
61 if (type == EventType::GameContextChanged)
62 {
63 // If we are getting into runtime
64 auto contextEvent = dynamic_cast<GameContextEvent&>(event);
65 if (contextEvent.GetNewState() == GameContextState::Runtime)
66 {
67 m_scene->OnAwake();
