276 template<typename Class, typename FieldType>
277 static void AddPropertyDirect(
278 FieldType Class::* member,
279 const std::string& name,
280 std::initializer_list<FieldOption> options,
281 std::vector<TypeField>& targetStack
282 ) {
283 using CleanedFieldT = std::decay_t<FieldType>;
284 constexpr EngineDataType type = DeduceEngineDataType<CleanedFieldT>();
285
286 FieldMeta meta;
287 std::string cleanedName = GetCleanedName(name);
288 meta.displayName = cleanedName;
289 for (auto&& opt : options) {
290 if (std::holds_alternative<FlattenTag>(opt)) { meta.flatten = true; }
291 else if (std::holds_alternative<DisplayNameTag>(opt)) { meta.displayName = std::get<DisplayNameTag>(opt).value; }
292 else if (std::holds_alternative<TooltipTag>(opt)) { meta.tooltip = std::get<TooltipTag>(opt).text; }
293 else if (std::holds_alternative<ReadOnlyTag>(opt)) { meta.readOnly = true; }
294 else if (std::holds_alternative<ColorTag>(opt)) { meta.isColor = true; }
295 }
296
297 alignas(Class) char dummyBuffer[sizeof(Class)];
298 Class* dummyObj = reinterpret_cast<Class*>(dummyBuffer);
299 size_t offset = static_cast<size_t>(
300 reinterpret_cast<const char*>(&(dummyObj->*member)) - dummyBuffer
301 );
302
303 TypeField field;
304
305 if constexpr (is_engine_array_v<CleanedFieldT>)
306 {
307 field = TypeField(name, EngineDataType::Array, offset, meta.displayName);
308
309 using ElementType = typename CleanedFieldT::value_type;
310 field.elementType = DeduceEngineDataType<ElementType>();
311
312 if constexpr (is_ee_object_ptr_v<ElementType>)
313 {
314 using TargetT = ee_ptr_target_t<ElementType>;
315 field.targetTypeKey = GetTypeKey<TargetT>();
316 }
317 else if constexpr (std::is_class_v<ElementType> && !std::is_same_v<ElementType, std::string>)
318 {
319 auto& customFields = GetReflectedTypes();
320 std::type_index ti = typeid(ElementType);
321 auto it = customFields.find(ti);
322 if (it != customFields.end())
323 {
324 field.elementChildren = it->second;
325 }
326 }
327
328 field.GetArraySize = [](const void* vecPtr) -> size_t {
329 if (!vecPtr) return 0;
330 const auto* vec = static_cast<const CleanedFieldT*>(vecPtr);
331 return vec->size();
332 };
333
334 field.GetElementAddress = [](const void* vecPtr, size_t index) -> const void* {
335 if (!vecPtr) return nullptr;
336 const auto* vec = static_cast<const CleanedFieldT*>(vecPtr);
337 if (index >= vec->size()) return nullptr;
338 return static_cast<const void*>(&(*vec)[index]);
339 };
340
341 field.ResizeArray = [](void* vecPtr, size_t newSize) {
342 if (!vecPtr) return;
343 auto* vec = static_cast<CleanedFieldT*>(vecPtr);
344 vec->resize(newSize);
345 };
346 }
347 else if constexpr (is_ee_object_ptr_v<CleanedFieldT>)
348 {
349 field = TypeField(name, EngineDataType::ObjectPtr, offset, meta.displayName);
350 using TargetT = ee_ptr_target_t<CleanedFieldT>;
351 field.targetTypeKey = GetTypeKey<TargetT>();
352 }
353 else if constexpr (is_unique_ptr_v<CleanedFieldT>)
354 {
355 using TargetT = unique_ptr_target_t<CleanedFieldT>;
356
357 std::vector<TypeField> subFields;
358 auto& customFields = GetReflectedTypes();
359
360 auto it = customFields.find(typeid(TargetT));
361 if (it != customFields.end())
362 {
363 subFields = it->second;
364 }
365
366 field = TypeField(name, EngineDataType::Custom, offset, meta.displayName, subFields);
367 field.targetTypeKey = GetTypeKey<TargetT>();
368
369 field.GetPointee = [](const void* ptr) -> const void*
370 {
371 return static_cast<const CleanedFieldT*>(ptr)->get();
372 };
373
374 field.GetPointeeTypeIndex = [](const void* ptr) -> std::type_index
375 {
376 const auto* pointee = static_cast<const CleanedFieldT*>(ptr)->get();
377
378 if (!pointee)
379 {
380 return typeid(void);
381 }
382
383 return pointee->GetTypeIndex();
384 };
385 }
386 else if (type == EngineDataType::Enum)
387 {
388 EnumRegistrar<CleanedFieldT>::Ensure();
389
390 field = TypeField(name, type, offset, meta.displayName);
391 field.targetTypeKey = GetTypeKey<CleanedFieldT>();
392 }
393 else if (type == EngineDataType::Custom)
394 {
395 auto& customFields = GetReflectedTypes();
396 std::type_index ti = typeid(CleanedFieldT);
397
398 std::vector<TypeField> subFields;
399 auto it = customFields.find(ti);
400 if (it != customFields.end()) {
401 subFields = it->second;
402 }
403 field = TypeField(name, EngineDataType::Custom, offset, meta.displayName, subFields);
404 }
405 else
406 {
407 field = TypeField(name, type, offset, meta.displayName);
408 }
409
410 field.flatten = meta.flatten;
411 field.isColor = meta.isColor;
412 field.tooltip = meta.tooltip;
413 field.readOnly = meta.readOnly;
414
415 targetStack.push_back(field);
416 }
