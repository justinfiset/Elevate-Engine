164 T& GameObject::AddComponent(Args&&... args)
165 {
166 static_assert(std::is_base_of_v<Component, T>, "AddComponent: T must derive from Component.");
167
168 auto& registry = Registry();
169 const uint32_t registryId = RegistryId();
170
171 if (T* existing = registry.TryGet<T>(registryId, m_entityId))
172 {
173 CoreLogger::Error("Error: Tried to add an already existing component to the {} GameObject", m_name);
174 return *existing;
175 }
176
177 T& comp = registry.Add<T>(registryId, m_entityId, std::forward<Args>(args)...);
178 comp.gameObject = this;
179 comp.Init();
180 return comp;
181 }
