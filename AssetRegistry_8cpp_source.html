21 std::string FormatAssetNameForUI(const std::filesystem::path& filePath)
22 {
23 std::string rawName = filePath.stem().string();
24
25 std::string result;
26 result.reserve(rawName.size());
27
28 bool capitalizeNext = true;
29 for (char c : rawName)
30 {
31 if (c == '_' || c == '-')
32 {
33 result += ' ';
34 capitalizeNext = true;
35 }
36 else
37 {
38 if (capitalizeNext && std::isalpha(static_cast<unsigned char>(c)))
39 {
40 result += static_cast<char>(std::toupper(static_cast<unsigned char>(c)));
41 capitalizeNext = false;
42 }
43 else
44 {
45 result += c;
46 }
47 }
48 }
49
50 return result;
51 }

58 void AssetRegistry::RefreshFromDisk()
59 {
60 fs::path rootPath = EE_CONTENT_ROOT;
61 if (!fs::exists(rootPath) || !fs::is_directory(rootPath))
62 {
63 EE_CORE_ERROR("Content root dir could not be resolved by Asset Registry.");
64 }
65
66 auto options = fs::directory_options::skip_permission_denied;
67
68 std::error_code ec;
69 for (const auto& entry : fs::recursive_directory_iterator(rootPath, options, ec))
70 {
71 if (!entry.is_regular_file())
72 {
73 continue;
74 }
75
76 std::string extension = entry.path().extension().string();
77 std::transform(extension.begin(), extension.end(), extension.begin(), ::tolower);
78
79 auto it = m_extensionMeta.find(extension);
80 if (it != m_extensionMeta.end())
81 {
82 const AssetMetaData& typeMeta = it->second;
83 Guid guid = ExtractGuidFromFile(entry.path());
84 s_indexedAssets[guid] = AssetEntry{
85 .AssetGuid = guid,
86 .AssetName = FormatAssetNameForUI(entry),
87 .FilePath = entry.path(),
88 .TypeIndex = typeMeta.TypeIndex,
89 .isOnDisk = true,
90 .isLoaded = false,
91 .MetaData = &typeMeta,
92 .Instance = nullptr
93 };
94
95 fs::path normalizedPath = fs::absolute(entry.path()).lexically_normal();
96 s_pathRegistry[normalizedPath] = guid;
97 }

130 Guid AssetRegistry::ExtractGuidFromFile(std::filesystem::path path)
131 {
132 FILE* fp = fopen(path.string().c_str(), "r");
133 if (!fp)
134 {
135 EE_CORE_ERROR("Cannot open Asset JSON file : {}", path.string());
136 return Guid{};
137 }
138
139 char readBuffer[65536];
140 rapidjson::FileReadStream is(fp, readBuffer, sizeof(readBuffer));
141
142 rapidjson::Document doc;
143 doc.ParseStream(is);
144 fclose(fp);
145
146 if (doc.HasParseError())
147 {
148 EE_CORE_ERROR("Erreur parsing JSON : %s", rapidjson::GetParseError_En(doc.GetParseError()));
149 return Guid{};
150 }
151
152 if (doc.HasMember(EEObject::GuidFieldName) && doc[EEObject::GuidFieldName].IsString())
153 {
154 std::string res = doc[EEObject::GuidFieldName].GetString();
155 EE_CORE_TRACE("Found asset with GUID : {}", res);
156 return Guid::FromString(res);
157 }
158 else
159 {
160 EE_CORE_ERROR("Could not find valid guid key in JSON file.");
161 }
162
163 return Guid{};
164 }
