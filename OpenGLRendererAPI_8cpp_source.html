72 void OpenGLRendererAPI::DrawArray(const VertexArray* vao, DrawPrimitiveType primitive) const
73 {
74 if (!glfwGetCurrentContext()) {
75 EE_CORE_ERROR("No OpenGL context active!");
76 return;
77 }
78
79 if (!vao)
80 {
81 EE_CORE_ERROR("VAO or IndexBuffer is null!");
82 return;
83 }
84
85 vao->Bind();
86
87 if (vao->GetIndexBuffer())
88 {
89 if (vao->GetIndexBuffer()->GetCount() == 0)
90 {
91 EE_CORE_ERROR("IndexBuffer has 0 indices!");
92 return;
93 }
94
95 GLCheck(glDrawElements(DrawPrimitiveTypeToOpenGL(primitive), vao->GetIndexBuffer()->GetCount(), GL_UNSIGNED_INT, nullptr));
96 }
97 else
98 {
99 for (const auto& vbo : vao->GetVertexBuffers())
100 {
101 uint32_t vertexCount = vbo->GetSize() / vbo->GetLayout().GetStride();
102 GLCheck(glDrawArrays(DrawPrimitiveTypeToOpenGL(primitive), 0, vertexCount));
103 }
104 }
