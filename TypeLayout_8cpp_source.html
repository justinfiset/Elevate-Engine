60 void SetPropertyRawValue(const TypeField& field, PropertyField& prop, uint16_t currentDepth)
61 {
62 if (field.type == EngineDataType::Array)

64 prop.Value = PropertyContainer{ CreateArrayPropertySet(field, prop.Path, currentDepth + 1) };

98 PropertySet CreateArrayPropertySet(const TypeField& parent, const std::string& parentPath, uint16_t currentDepth)
99 {
100 PropertySet arraySet;
101
102 if (!parent.data || !parent.GetArraySize || !parent.GetElementAddress)
103 return arraySet;
104

146 PropertySet CreateContainer(const TypeField& parent, const std::string& parentPath, uint16_t currentDepth)
147 {
148 PropertySet set;
