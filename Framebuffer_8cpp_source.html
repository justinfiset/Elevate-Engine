23 Framebuffer* Framebuffer::Create(uint32_t width, uint32_t height, std::vector<TextureFormat> colorFormats, bool depthAsRenderbuffer, TextureType depthUsage)
24 {
25 if (width == 0 || height == 0)
26 {
27 EE_CORE_WARN("Framebuffer::Create - Attempted to create framebuffer with 0x0 dimensions. Skipping.");
28 return nullptr;
29 }
30
31 size_t colorAttachmentCount = colorFormats.size();
32
33 std::vector<TexturePtr> colorTextures;
34 colorTextures.reserve(colorAttachmentCount);
35 for (uint32_t i = 0; i < colorAttachmentCount; i++)
36 {
37 colorTextures.push_back(CreateColorTexture(width, height, colorFormats[i]));
38 }
39
40 TexturePtr depthTex = nullptr;
41 if (!depthAsRenderbuffer)
42 {
43 depthTex = CreateDepthTexture(width, height, TextureFormat::DEPTH, depthUsage);
44 }
45 Framebuffer* buffer = Framebuffer::Create(colorTextures, depthTex, depthAsRenderbuffer);
46 buffer->m_width = width;
47 buffer->m_height = height;
48 return buffer;
49 }
